import type { CSSProperties } from "react"

export interface RadioOption {
  value: string
  label: string
  description?: string
  /** Right-aligned trailing text — a price in the card variant. */
  meta?: string
  disabled?: boolean
}

/**
 * One choice from a small set. The `card` variant is the shipping- and payment-method picker.
 * @startingPoint section="Forms" subtitle="Radio list and selectable cards" viewport="700x260"
 */
export interface RadioGroupProps {
  name?: string
  options: RadioOption[]
  value?: string
  defaultValue?: string
  onChange?: (value: string) => void
  variant?: "list" | "card"
  style?: CSSProperties
}

export declare function RadioGroup(props: RadioGroupProps): JSX.Element
